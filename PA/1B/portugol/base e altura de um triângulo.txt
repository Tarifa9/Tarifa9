programa {
  funcao inicio() {

    real base, altura, area

    escreva("digite o valor da base do triangulo: ")
    leia(base)
    escreva("digite o valor da altura do triangulo")
    leia(altura)

    area = (base * altura) / 2

    escreva("\n---Dados do Triângulo---")
    escreva("\nbase: ", base)
    escreva("\naltura: ", altura)
    escreva("\na area do triângulo é: ", area)
  }
}
