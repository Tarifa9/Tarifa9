* Descrição: Ler uma base menor, uma base maior e uma altura. Calcular e mostrar a

programa {
  funcao inicio() {

    real baseMenor, baseMaior, altura, area

    escreva("digite o valor da base menor: ")
    leia(baseMenor)
    escreva("digite o valor da base maior: ")
    leia(baseMaior)
    escreva("digite o valor de altura: ")
    leia(altura)

    area = ((baseMenor + baseMaior) * altura) / 2

    escreva("\n--- Cálculo do Trapézio ---")
		escreva("\nBase Menor: ", baseMenor)
		escreva("\nBase Maior: ", baseMaior)
		escreva("\nAltura: ", altura)
		escreva("\nA área do trapézio é: ", area)
	}
}
