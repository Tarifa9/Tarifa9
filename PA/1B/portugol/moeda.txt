* Calcular e exibir o valor recebido de cada um dos tipos de moeda e o soma total em moedas.

programa {
  funcao inicio() {

    inteiro q1=0, q5=0, q10=0, q25=0, q50=0, q1R=0
    real v1=0.0, v5=0.0, v10=0.0, v25=0.0, v50=0.0, v1R=0.0, total=0.0

    escreva("Qtd moedas de 1 centavo: ")
    leia(q1)
    escreva("Qtd moedas de 5 centavo: ")
    leia(q5)
    escreva("Qtd moedas de 10 centavo: ")
    leia(q10)
    escreva("Qtd moedas de 25 centavo: ")
    leia(q25)
    escreva("Qtd moedas de 50 centavos: ")
    leia(q50)
    escreva("Qtd moedas de 1 real: ")
    (q1R)

    v1 = q1 * 0.01
    v5 = q5 * 0.05
    v10 = q10 * 0.10
    v25 = q25 * 0.25
    v50 = q50 * 0.50
    v1R = q1R * 1.00

    total = v1 + v5 + v10 + v25 + v50 + v1R

    escreva("\n--- Resumo do Caixa ---")
		escreva("\n1 centavo : R$ ", v1)
		escreva("\n5 centavos: R$ ", v5)
		escreva("\n10 centavos: R$ ", v10)
		escreva("\n25 centavos: R$ ", v25)
		escreva("\n50 centavos: R$ ", v50)
		escreva("\n1 real     : R$ ", v1R)
		escreva("\nTOTAL GERAL: R$ ", total)
  }
}
