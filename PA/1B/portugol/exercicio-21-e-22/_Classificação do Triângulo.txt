*
* Equilátero (todos os lados iguais)
* Isósceles (dois lados iguais)
* Escaleno (todos os lados diferentes).
*/

programa {
  funcao inicio() {
    real a, b, c

		escreva("Digite o lado A: ")
		leia(a)
		escreva("Digite o lado B: ")
		leia(b)
		escreva("Digite o lado C: ")
		leia(c)

		se (a < b + c e b < a + c e c < a + b)
		{
			escreva("Os lados formam um triângulo: ")

			se (a == b e b == c)
			{
				escreva("Equilátero (todos os lados iguais)")
			}
			senao se (a == b ou a == c ou b == c)
			{
				escreva("Isósceles (dois lados iguais)")
			}
			senao
			{
				escreva("Escaleno (todos os lados diferentes)")
			}
		}
		senao
		{
			escreva("Os valores informados NÃO podem formar um triângulo.")
		
	}
}
  }
}
