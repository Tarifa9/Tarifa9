* Descrição: Ler o peso de um boi e o percentual de engorda. Calcular e exibir o novo peso do boi.
*/

programa {
  funcao inicio() {

    real pesoAtual, percentualEngorda, aumentoPeso, novoPeso

    escreva("digitre o peso atual do boi (kg): ")
    leia(pesoAtual)
    escreva("digite o percentual de engorda (exemplos: 15 para 15%): ")
    leia(percentualEngorda)
    
    aumentoPeso = pesoAtual * (percentualEngorda / 100)
    novoPeso = pesoAtual + aumentoPeso

    escreva("\n---Relatório de Engorda---")
		escreva("\nPeso Original: ", pesoAtual, " kg")
		escreva("\nPercentual de Engorda: ", percentualEngorda, "%")
		escreva("\nGanho de Peso: ", aumentoPeso, " kg")
    escreva("\nNovo Peso do Boi: ", novoPeso, " kg")
  }
}
