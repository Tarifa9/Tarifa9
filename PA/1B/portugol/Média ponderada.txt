programa {
  funcao inicio() {

    real n1, n2, n3, n4, media

    escreva("digite a primeira nota (peso 1): ")
    leia(n1)
    escreva("digite a segunda nota (peso 2): ")
    leia(n2)
    escreva("digite a terceira nota (peso 3): ")
    leia(n3)
    escreva("digite a quarta no (peso 4): ")
    leia(n4)

    media = ((n1 * 1) + (n2 * 2) + (n3 * 3) + (n4 * 4)) / 10

    escreva("\n---Notas Digitadas---")
		escreva("\nNota 1: ", n1, " | Nota 2: ", n2)
		escreva("\nNota 3: ", n3, " | Nota 4: ", n4)
		escreva("\nMédia Ponderada: ", media)

  }
}
